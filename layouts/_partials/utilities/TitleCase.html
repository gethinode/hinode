{{- /* Applies title case to a string, honoring Hugo's `titleCaseStyle` setting. Accepts rendered HTML: tags and character entities pass through untouched, so inline markup such as `` from a markdown backtick survives intact. The casing is applied once to the concatenated text content and then mapped back onto the original by rune position, which keeps stopword handling correct across tag boundaries. Character entities contribute a single space to the plain string used for casing, rather than nothing. Omitting them entirely would join the words on either side of the entity, so the word right after it would never be treated as starting a word (e.g. `the cat sat` would case to `The cat Sat` instead of `The Cat Sat`). The entity itself is still always re-emitted verbatim in the output — only the *casing pass* sees a space stand-in for it — and because a space cases to a space, the one-rune width keeps the positional mapping back onto the original tokens exact. Words listed in `site.Params.main.titleCaseExceptions` keep their exact spelling — for lowercase-authored acronyms such as `npm`, which `title` would otherwise capitalize to `Npm`. Each entry is matched as a whole word against its own title-cased form (`\btitle($entry)\b`), so only that exact casing is restored; a deliberately shouted spelling such as `NPM` does not match `\bNpm\b` and is left alone. Entries must be plain words: Go's regexp has no `\Q…\E` escaping, so an entry containing regex metacharacters would misbehave. @param {string} text Plain string or rendered HTML. @param {page} page Page supplying `.Params.exact`. Optional when `exact` is passed. @param {bool} exact Optional. Overrides `page.Params.exact` for callers holding no page. @returns {string} The title-cased text, or the input unchanged when title casing is disabled, opted out of, or cannot be applied safely. */ -}} {{- $text := .text -}} {{- $result := $text -}} {{- $exceptions := site.Params.main.titleCaseExceptions | default slice -}} {{- $exact := false -}} {{- if isset . "exact" -}} {{- $exact = .exact -}} {{- else -}} {{- with .page -}}{{- $exact = .Params.exact -}}{{- end -}} {{- end -}} {{- if and site.Params.main.titleCase (not $exact) $text -}} {{- if not (or (strings.Contains $text "<") (strings.Contains $text "&")) -}} {{/* Fast path: no markup and no entities, so the filter can be applied directly. */}} {{- $result = title $text -}} {{- range $exceptions -}} {{- $result = replaceRE (printf `\b%s\b` (title .)) . $result -}} {{- end -}} {{- else -}} {{/* Split into tags, character entities, and text runs. Only the text runs are cased. */}} {{- $tokens := findRE `<[^>]+>|&#?[0-9a-zA-Z]+;|[^<&]+|[<&]` $text -}} {{- $plain := slice -}} {{- range $tokens -}} {{- if and (hasPrefix . "<") (hasSuffix . ">") -}} {{- else if and (hasPrefix . "&") (hasSuffix . ";") -}} {{- $plain = $plain | append " " -}} {{- else -}} {{- $plain = $plain | append . -}} {{- end -}} {{- end -}} {{- $plain = delimit $plain "" -}} {{- $cased := title $plain -}} {{- range $exceptions -}} {{- $cased = replaceRE (printf `\b%s\b` (title .)) . $cased -}} {{- end -}} {{- $runes := split $cased "" -}} {{/* The filter only changes case, so lengths must match. If they ever do not, the positional mapping would corrupt the markup — return the input untouched. */}} {{- if eq (len $runes) (len (split $plain "")) -}} {{- $out := slice -}} {{- $pos := 0 -}} {{- range $tokens -}} {{- if and (hasPrefix . "<") (hasSuffix . ">") -}} {{- $out = $out | append . -}} {{- else if and (hasPrefix . "&") (hasSuffix . ";") -}} {{- $out = $out | append . -}} {{- $pos = add $pos 1 -}} {{- else -}} {{- $n := len (split . "") -}} {{- $out = $out | append (delimit (first $n (after $pos $runes)) "") -}} {{- $pos = add $pos $n -}} {{- end -}} {{- end -}} {{- $result = delimit $out "" -}} {{- end -}} {{- end -}} {{- end -}} {{- return $result -}}
