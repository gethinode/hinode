{{- /* Resolves a boolean page parameter, checking the page's front matter, then the per-type site parameter, then the global site parameter, then the caller's default. Mirrors GetIncludeTOC.html, which resolves `includeToc` the same way. Site parameter keys are lowercased by Hugo, so the name is lowercased before every lookup. @param {page} page Page to resolve for. @param {string} name Parameter name, e.g. "readingTime". @param {bool} default Value to use when nothing else sets it. @returns {bool} */ -}} {{- $page := .page -}} {{- $name := lower .name -}} {{- $value := .default -}} {{- $source := printf "page parameters: %s (%s)" .name $page.RelPermalink -}} {{- with site.Params.pages -}} {{- if isset . $name -}} {{- $value = index . $name -}} {{- $source = printf "site parameters: pages.%s" $.name -}} {{- end -}} {{- end -}} {{- with (index site.Params.pages $page.Type) -}} {{- if isset . $name -}} {{- $value = index . $name -}} {{- $source = printf "site parameters: pages.%s.%s" $page.Type $.name -}} {{- end -}} {{- end -}} {{- if isset $page.Params $name -}} {{- $value = index $page.Params $name -}} {{- $source = printf "page parameters: %s (%s)" $.name $page.RelPermalink -}} {{- end -}} {{- if ne (printf "%T" $value) "bool" -}} {{- errorf "Expected bool value in %s" $source -}} {{- end -}} {{- return $value -}}
