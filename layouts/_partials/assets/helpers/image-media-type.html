The extension test is case-insensitive and ignores a query string or fragment. Returns an empty string for an extension not listed below, so that a caller emits no type rather than a guessed one. "src" Path or URL of the image. */}} {{- $types := dict "png" "image/png" "jpg" "image/jpeg" "jpeg" "image/jpeg" "gif" "image/gif" "webp" "image/webp" "avif" "image/avif" "svg" "image/svg+xml" -}} {{- $ext := lower (strings.TrimPrefix "." (path.Ext (urls.Parse (.src | default "")).Path)) -}} {{- return (index $types $ext | default "") -}}
