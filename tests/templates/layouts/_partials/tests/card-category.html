{{- /* The category header reads as a quiet sentence-case label: no uppercase transform, an abbreviated month, a margin rather than a bullet between category and date, and the same muted ink on both. Other header styles keep the uppercase transform. */ -}} {{- $target := site.GetPage "/blog/card-category" -}} {{- $render := partial "assets/card.html" (dict "page" $target "path" $target.RelPermalink "header-style" "category" "footer-style" "none" ) -}} {{- $html := string $render -}} {{- range slice `Announcements` (printf `%s` ($target.Date | time.Format ":date_medium")) -}} {{- if not (strings.Contains $html .) -}} {{- errorf "card category: missing %q in %s" . $html -}} {{- end -}} {{- end -}} {{- range slice "text-uppercase" "•" "•" "October" -}} {{- if strings.Contains $html . -}} {{- errorf "card category: unexpected %q in %s" . $html -}} {{- end -}} {{- end -}} {{- $date := string (partial "assets/card.html" (dict "page" $target "path" $target.RelPermalink "header-style" "date" "footer-style" "none" )) -}} {{- if not (strings.Contains $date "text-uppercase") -}} {{- errorf "card category: date header lost its uppercase transform in %s" $date -}} {{- end -}}
